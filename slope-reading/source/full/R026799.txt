"""Refresh the kit PACKAGE_MANIFEST.json in the current working directory's kit: update changed files, add new ones."""
import hashlib
import json
from pathlib import Path

kit = Path.cwd() / 'Slope_Credit_Scenario_Research_and_Design_Kit'
p = kit / 'PACKAGE_MANIFEST.json'
raw = p.read_text()
m = json.loads(raw)
known = {f['path'] for f in m['files']}
changed, added = [], []
for f in m['files']:
    fp = kit / f['path']
    if not fp.exists():
        print('missing', f['path'])
        continue
    data = fp.read_bytes()
    sha, size = hashlib.sha256(data).hexdigest(), len(data)
    if (sha, size) != (f['sha256'], f['bytes']):
        f['sha256'], f['bytes'] = sha, size
        changed.append(f['path'])
for fp in sorted((kit / 'research/recent_cases').rglob('*')):
    rel = str(fp.relative_to(kit))
    if fp.is_file() and rel not in known and not fp.name.startswith('.'):
        data = fp.read_bytes()
        m['files'].append({'path': rel, 'bytes': len(data), 'sha256': hashlib.sha256(data).hexdigest()})
        added.append(rel)
m['files'].sort(key=lambda f: f['path'])
indent = 2 if raw.startswith('{\n  "') else None
p.write_text(json.dumps(m, indent=indent, ensure_ascii=False) + ('\n' if raw.endswith('\n') else ''))
