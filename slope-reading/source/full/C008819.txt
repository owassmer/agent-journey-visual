A worker only runs a walker while it holds a task, and it re-reads the\n# queue after every task, so zero walkers twice means no task was open for it to claim. Exits when both are stopped.\ncd ~/.hermes/profiles/connor/cache/scratch/split\nA=\"/opt/homebrew/bin/aws --profile slope --region us-east-2\"\nP=s3://slope-walk-462947327980-20261001/walk-36781427817/fresh-corrected-3\ndeclare -A idle\nHOSTS=\"i-070d4380d338ef0af i-04dffa9a231480f14\"\nsleep 600  # let both workers claim their first tasks\nwhile true; do\n  left=0\n  for i in $HOSTS; do\n    state=$($A ec2 describe-instances --instance-ids $i --query 'Reservations[0].Instances[0].State.Name' --output text)\n    [ \"$state\" != \"running\" ] && continue\n    left=$((left + 1))\n    n=$(python3 ssm_$i.py --wait 30 -- \"pgrep -fc app.disputes.parallel\" 2>/dev/null | head -1)\n    saved=$($A s3 ls $P/refine-v1/done/ 2>/dev/null | grep -c json)\n    echo \"$(date -u +%H:%M) $i walkers ${n:-?} tasks saved $saved/1050\" >> stop_big.log\n    if [ \"$n\" = \"0\" ]; then\n      idle[$i]=$(( ${idle[$i]:-0} + 1 ))\n      if [ \"${idle[$i]}\" -ge 2 ]; then\n        $A ec2 stop-instances --instance-ids $i --query 'StoppingInstances[0].CurrentState.Name' --output text\n        echo \"$(date -u +%H:%M) STOPPED $i: no task to claim on two checks; tasks saved $saved/1050\" | tee -a stop_big.log\n      fi\n    else\n      idle[$i]=0\n    fi\n  done\n  [ \"$left\" -eq 0 ] && break\n  sleep 180\ndone\necho \"both large hosts stopped\"\n$A ec2 describe-instances --instance-ids $HOSTS --query 'Reservations[].Instances[].[InstanceId,State.Name]' --output text\n"}
