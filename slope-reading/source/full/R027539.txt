1	"""Present-state interpretation of one live dispute: Jev reads each cited finding, hydrated with its surrounding
2	evidence, against one obligation; code derives the structural facts.
3	
4	Jev returns full distributions (Choice, Noul, Score), all kept. Code derives only what is structural: who pays (the
5	readings must agree), the amount's status (the most advanced established), which procedural events are established
6	(and so the stage), and the constraints an established fact imposes (a waiver covering this obligation removes the
7	appeal branch; payment resolves the dispute). Factor distributions are aggregated by each factor's rule and passed to
8	the forecasts as context; they never become probabilities themselves.
9	"""
10	
11	from __future__ import annotations
12	
13	import asyncio
14	from collections.abc import Callable
15	from typing import Protocol
16	
17	from app.disputes.rules import load_model
18	from app.domain.investigation import (
19	    AtomicFinding,
20	    Decisive,
21	    DisputeInstance,
22	    EvidenceRequest,
23	    FactorResult,
24	    FindingReading,
25	    SemanticObservation,
26	)
27	
28	ESTABLISHED = 0.5  # code-owned: a Noul value at or above this establishes a present-state fact
29	DATED_STAGES = ("judgment_entered", "appeal_pending", "enforcement")
30	QUOTE_CHARS = 300
31	
32	
33	class InterpretationJudge(Protocol):
34	    async def read(self, obligation: dict, evidence: dict, direction: dict[str, str],
35	                   subject_ids: tuple[str, ...]) -> list[SemanticObservation]: ...
36	
37	    async def relevance(self, obligation: dict, evidence: dict, subject_ids: tuple[str, ...]) -> list[SemanticObservation]: ...
38	
39	    async def level(self, obligation: dict, evidence: dict, factor: dict, rubric: list[str],
40	                    subject_ids: tuple[str, ...]) -> SemanticObservation: ...
41	
42	
43	def argmax(d: dict[str, float]) -> str | None:
44	    return max(d, key=lambda k: d[k]) if d else None
45	
46	
47	def expected_level(dist: dict[str, float], levels: list[str]) -> float:
48	    return sum(levels.index(k) * p for k, p in dist.items() if k in levels)
49	
50	
51	def obligation_state(inst: DisputeInstance, borrower: str, model: dict) -> dict:
52	    """What Jev is told about the obligation: a docket reference, its nature and the two parties, unordered.
53	    Nothing the agent wrote about who pays, the amount's status or anyone's intentions."""
54	    return {"reference": inst.order_reference, "nature": model["natures"].get(inst.nature, inst.nature),
55	            "parties": sorted([borrower, inst.counterparty])}
56	
57	
58	class Interpreter:
59	    def __init__(self, draft: DisputeInstance, findings: list[AtomicFinding], *, judge: InterpretationJudge | None,
60	                 borrower: str, sources: dict[str, tuple[str, str]], hydrate: Callable[[AtomicFinding], dict],
61	                 agent_reading: dict | None = None, model: dict | None = None) -> None:
62	        self.m = model or load_model()
63	        self.d, self.findings, self.judge = draft, findings, judge
64	        self.borrower, self.sources, self.hydrate = borrower, sources, hydrate
65	        self.agent_reading = agent_reading or {}
66	        self.obligation = obligation_state(draft, borrower, self.m)
67	        self.obs: list[str] = []
68	        self.requests: list[EvidenceRequest] = []
69	
70	    def _date(self, f: AtomicFinding) -> str:
71	        return max(self.sources.get(s.source_id, ("", ""))[1] for s in f.spans)
72	
73	    def _decisive(self, f: AtomicFinding) -> Decisive:
74	        quote = " … ".join(s.quote for s in f.spans)
75	        return Decisive(finding_id=f.finding_id, source_date=self._date(f),
76	                        quote=quote if len(quote) <= QUOTE_CHARS else quote[:QUOTE_CHARS] + "…")
77	
78	    # --- Jev's readings ---------------------------------------------------------------------------
79	
80	    async def _read(self, f: AtomicFinding, evidence: dict) -> FindingReading:
81	        direction = {"borrower": f"{self.borrower} must pay {self.d.counterparty}",
82	                     "counterparty": f"{self.d.counterparty} must pay {self.borrower}",
83	                     "not_stated": "The evidence does not say which party pays under this obligation"}
84	        obs = await self.judge.read(self.obligation, evidence, direction, (f.finding_id, self.d.instance_id))
85	        self.obs += [o.observation_id for o in obs]
86	        by = {o.question_id: o for o in obs}
87	        return FindingReading(
88	            finding_id=f.finding_id, source_date=self._date(f),
89	            payer=dict(by["obligation_direction"].probabilities or {}) if "obligation_direction" in by else {},
90	            amount_status=dict(by["amount_status"].probabilities or {}) if "amount_status" in by else {},
91	            includes_interest=by["amount_includes_interest"].noul_value if "amount_includes_interest" in by else None,
92	            events={ev: (by[f"event_{ev}"].noul_value if f"event_{ev}" in by else None) for ev in self.m["readings"]["events"]},
93	            observation_ids=tuple(o.observation_id for o in obs))
94	
95	    async def _factors(self, f: AtomicFinding, r: FindingReading, evidence: dict, obligation: dict) -> FindingReading:
96	        obs = await self.judge.relevance(obligation, evidence, (f.finding_id, self.d.instance_id))
97	        self.obs += [o.observation_id for o in obs]
98	        by = {o.question_id: o for o in obs}
99	        bears = {fid: (by[f"bears_on_{fid}"].noul_value if f"bears_on_{fid}" in by else None) for fid in self.m["factors"]}
100	        levels, oids = {}, [o.observation_id for o in obs]
101	        for fid, spec in self.m["factors"].items():
102	            if spec["kind"] == "graded" and (bears.get(fid) or 0) >= ESTABLISHED:
103	                factor = {"factor_id": fid, "label": spec["label"], "question": spec["relevance"]}
104	                o = await self.judge.level(obligation, evidence, factor, spec["levels"], (f.finding_id, self.d.instance_id))
105	                self.obs.append(o.observation_id)
106	                oids.append(o.observation_id)
107	                levels[fid] = {spec["levels"][int(k)]: v for k, v in (o.probabilities or {}).items()
108	                               if str(k).isdigit() and int(k) < len(spec["levels"])}
109	        return r.model_copy(update={"bears_on": bears, "levels": levels,
110	                                    "observation_ids": r.observation_ids + tuple(oids)})
111	
112	    # --- aggregation (code) -----------------------------------------------------------------------
113	
114	    def _aggregate(self, readings: list[FindingReading]) -> list[FactorResult]:
115	        byid = {f.finding_id: f for f in self.findings}
116	        out = []
117	        for fid, spec in self.m["factors"].items():
118	            base = {"factor_id": fid, "label": spec["label"], "kind": spec["kind"], "aggregate": spec["aggregate"]}
119	            if spec["kind"] == "present":
120	                scored = [r for r in readings if r.bears_on.get(fid) is not None]
121	                if not scored:
122	                    out.append(FactorResult(**base))
123	                    continue
124	                top = max(scored, key=lambda r: r.bears_on[fid])
125	                p = top.bears_on[fid]
126	                out.append(FactorResult(**base, probability=p, level_label="established" if p >= ESTABLISHED else "not established",
127	                                        decisive=self._decisive(byid[top.finding_id]) if p >= ESTABLISHED else None,
128	                                        finding_ids=tuple(r.finding_id for r in scored if r.bears_on[fid] >= ESTABLISHED)))
129	                continue
130	            graded = [r for r in readings if r.levels.get(fid)]
131	            if not graded:
132	                out.append(FactorResult(**base))
133	                continue
134	            levels = spec["levels"]
135	            if spec["aggregate"] == "most_advanced":
136	                pick = [max(graded, key=lambda r: (expected_level(r.levels[fid], levels), r.source_date))]
137	            else:  # latest: the most recent passage(s); same-date disagreement is kept as a conflict
138	                newest = max(r.source_date for r in graded)
139	                pick = [r for r in graded if r.source_date == newest]
140	            dist = {lv: sum(r.levels[fid].get(lv, 0.0) for r in pick) / len(pick) for lv in levels}
141	            conflict = len({argmax(r.levels[fid]) for r in pick}) > 1
142	            out.append(FactorResult(**base, distribution=dist, conflict=conflict,
143	                                    level_label="conflicting readings" if conflict else argmax(dist),
144	                                    decisive=self._decisive(byid[pick[0].finding_id]),
145	                                    finding_ids=tuple(r.finding_id for r in graded)))
146	        return out
147	
148	    def _request(self, factor_id: str, action: str) -> None:
149	        self.requests.append(EvidenceRequest(factor_id=factor_id, action=action,
150	                                             if_satisfied="The dispute enters the analysis.",
151	                                             if_not="The analysis shows this dispute as not modelled."))
152	
153	    def _outside(self, update: dict) -> DisputeInstance:
154	        return self.d.model_copy(update={**update, "status": "outside_model", "evidence_requests": tuple(self.requests),
155	                                         "observation_ids": tuple(self.obs)})
156	
157	    # --- run --------------------------------------------------------------------------------------
158	
159	    async def run(self) -> DisputeInstance:
160	        update: dict = {"model_id": self.m["model_id"], "model_version": self.m["model_version"]}
161	        if self.judge is None:  # agent-only arm: the agent's own reading of the side and the stage
162	            role, stage = self.agent_reading.get("borrower_role"), self.agent_reading.get("stage")
163	            if role not in ("debtor", "creditor") or stage not in DATED_STAGES + ("amount_pending",):
164	                return self.d.model_copy(update={**update, "status": "not_judged"})
165	            return self.d.model_copy(update={**update, "borrower_role": role, "stage": stage, "status": "not_judged"})
166	        order = self.m["readings"]["amount_status_order"]
167	        evidence = {f.finding_id: self.hydrate(f) for f in self.findings}
168	        readings = list(await asyncio.gather(*(self._read(f, evidence[f.finding_id]) for f in self.findings)))
169	        payers = {argmax(r.payer) for r in readings} & {"borrower", "counterparty"}
170	        if len(payers) != 1:
171	            self._request("direction", f"Confirm from the order or judgment ({self.d.order_reference}) which party pays.")
172	            return self._outside({**update, "readings": tuple(readings)})
173	        role = "debtor" if payers == {"borrower"} else "creditor"
174	        obligation = {**self.obligation, "payer": self.borrower if role == "debtor" else self.d.counterparty}
175	        readings = list(await asyncio.gather(*(self._factors(f, r, evidence[f.finding_id], obligation)
176	                                               for f, r in zip(self.findings, readings, strict=True))))
177	        factors = self._aggregate(readings)
178	        statuses = [argmax(r.amount_status) for r in readings]
179	        statuses = [s for s in statuses if s in order]
180	        status = max(statuses, key=order.index) if statuses else "unknown"
181	        interest = any((r.includes_interest or 0) >= ESTABLISHED for r in readings
182	                       if argmax(r.amount_status) not in (None, "not_stated"))
183	        byid = {f.finding_id: f for f in self.findings}
184	        established = {}
185	        for ev in self.m["readings"]["events"]:
186	            yes = [r for r in readings if (r.events.get(ev) or 0) >= ESTABLISHED]
187	            if yes:
188	                established[ev] = self._decisive(byid[max(yes, key=lambda r: r.source_date).finding_id])
189	        stage = next((rule["stage"] for rule in self.m["readings"]["stage_rules"] if rule["event"] in established), None)
190	        if status == "fixed" and stage == "amount_pending":
191	            stage = "judgment_entered"
192	        constraints = {}
193	        waiver = next((f for f in factors if f.factor_id == "appeal_barred"), None)
194	        if waiver and waiver.probability is not None and waiver.probability >= ESTABLISHED and waiver.decisive:
195	            constraints["appeal"] = (f"An appeal of this obligation is waived or barred ({waiver.decisive.finding_id}: "
196	                                     f"“{waiver.decisive.quote}”)")
197	        update.update({"borrower_role": role, "amount_status": status, "amount_includes_interest": interest,
198	                       "readings": tuple(readings), "factors": tuple(factors), "established": established,
199	                       "stage": stage, "constraints": constraints})
200	        if stage == "paid" or status == "paid":
201	            return self.d.model_copy(update={**update, "status": "resolved", "observation_ids": tuple(self.obs)})
202	        if stage is None:
203	            self._request("stage", f"Obtain the docket for {self.d.order_reference} to establish where it stands.")
204	            return self._outside(update)
205	        if stage in DATED_STAGES and self.d.judgment_date is None:
206	            self._request("judgment_date", "Confirm the judgment's entry date from the docket.")
207	            return self._outside(update)
208	        return self.d.model_copy(update={**update, "status": "interpreted", "observation_ids": tuple(self.obs)})
