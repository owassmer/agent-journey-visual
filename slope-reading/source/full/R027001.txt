49	class StubJudge:
50	    """Answers each atomic question from fixed preferences; `top` is the transition answer's share."""
51	
52	    def __init__(self, stage: str, prefer: dict[str, str], top: float = 0.8):
53	        self.stage_id, self.prefer, self.top, self.calls = stage, prefer, top, []
54	
55	    def _obs(self, qid, options, pick, top, primitive="choice"):
56	        keys = list(options)
57	        rest = (1 - top) / (len(keys) - 1)
58	        probs = {k: (top if k == pick else rest) for k in keys}
59	        self.calls.append(qid)
60	        return SemanticObservation(observation_id=f"obs_{len(self.calls)}", call_id="c", profile="dispute_model",
61	                                   question_id=qid, question_version="3.10.0", primitive=primitive, answer=pick,
62	                                   probabilities=probs)
63	
64	    async def stage(self, dispute, record, options, subject_ids):
65	        return self._obs("dispute_stage", options, self.stage_id, 0.9)
66	
67	    async def route(self, dispute, finding, options, subject_ids):
68	        return self._obs("factor_routing", options, "appeal_intent", 0.9)
69	
70	    async def level(self, dispute, finding, factor, rubric, subject_ids):
71	        return self._obs("factor_level", {str(i): r for i, r in enumerate(rubric)}, "3", 0.9, "score")
72	
73	    async def present(self, dispute, finding, factor, subject_ids):
74	        return self._obs("factor_present", {"true": 1, "false": 0}, "false", 0.9)
75	
76	    async def transition(self, dispute, record, stage, premise, factor_results, options, subject_ids):
77	        top = 0.85 if factor_results else self.top
78	        return self._obs("dispute_transition", options, self.prefer.get(stage["stage"], next(iter(options))), top)

82	    f = finding("fnd_001", quote, source)
83	    ev_ = Evaluator(instance(role, amount, judgment, f.finding_id), [f], judge=judge, borrower="ChromaDex, Inc.",

85	    return asyncio.run(ev_.run())

90	def test_offers_conserve_totals_and_prorate_fees():
91	    terms = load_terms()
92	    for o in menu(terms, 200_000_000, "B"):
93	        sched = o.schedule(FUNDING)
94	        assert sum(p.amount_cents for p in sched) == o.total_cents
95	        assert sum(p.principal_cents for p in sched) == o.amount_cents
96	        assert all(p.due > FUNDING for p in sched)
97	        assert o.prorated_fee_cents(0) == 0 and o.prorated_fee_cents(o.days) == o.fee_cents
98	    assert offer(terms, "net_30", 200_000_000, "A").fee_cents == 3_200_000  # 1.60% on Net 30, the published rate

113	def test_the_invoice_is_paid_once_including_a_partly_financed_remainder():
114	    feed, terms = load_feed(SNAP), load_terms()
115	    declined, _, _, _ = simulate(feed, REQ, None, [], "central", HORIZON)
116	    for amount in (200_000_000, 120_000_000):  # the full invoice, and a smaller supported amount
117	        o = offer(terms, "net_30", amount, "A")
118	        financed, coll, owed, _ = simulate(feed, REQ, o, [], "central", HORIZON)
119	        assert declined[HORIZON] - financed[HORIZON] == o.fee_cents  # financing costs exactly the fee
120	        assert sum(c for _, c in coll) == o.total_cents and not owed

123	def test_a_lock_lowers_cash_by_its_amount_and_zero_cash_paths_reproduce_bank_only():
124	    feed, terms = load_feed(SNAP), load_terms()
125	    lock = BranchCash(kind="lock", label="bond", amount=ev(980_000_000), window_start=FUNDING,
126	                      window_end=date(2024, 12, 31))
127	    s = offer(terms, "inst_90", 200_000_000, "A")
128	    base, _, _, _ = simulate(feed, REQ, s, [], "stress", HORIZON)
129	    locked, coll, _, _ = simulate(feed, REQ, s, [lock], "stress", HORIZON)
130	    assert all(base[d] - locked[d] == 980_000_000 for d in base if d >= FUNDING)
131	    assert sum(c for _, c in coll) <= s.total_cents
132	    empty = evaluate("debtor", 1_000_000, date(2024, 8, 13), CA_QUOTE, "cacd_16cv2277_d618_judgment",
133	                     StubJudge("judgment_entered", {"judgment_entered": "appeal_bonded", "appeal_pending": "appeal_continues"}))
134	    no_cash = [p.model_copy(update={"cash": ()}) for p in empty.paths]
135	    comp = compare(feed, terms, REQ, [no_cash])
136	    for name in list(comp.structures):
137	        rows = {(o.min_cash_cents, tuple(o.collections)) for o in comp.outcomes if o.structure == name}
138	        assert len(rows) == 1

141	def test_the_supported_amount_is_sized_on_its_own_paths_and_names_what_binds():
142	    feed, terms = load_feed(SNAP), load_terms()
143	    bond = BranchCash(kind="lock", label="bond", amount=ev(1_225_000_000), window_start=FUNDING,
144	                      window_end=date(2024, 9, 30))
145	
146	    paths = [DisputePath(path_id="p1", transitions=("appeal_bonded",), labels=("bond",), terminal="t", weight_bps=10_000,
147	                         cash=(bond,))]
148	    comp = compare(feed, terms, REQ, [paths])
149	    s = summarize(comp, terms)
150	    rec = s["recommendation"]["event_adjusted"]
151	    assert not rec["requested_passes"] and not rec["at_requested"]["passes"] and rec["at_requested"]["lowest_cash_on"]
152	    assert rec["structure"] != "decline" and rec["at_recommended"]["passes"]
153	    chosen = comp.structures[rec["structure"]]
154	    assert chosen.amount_cents <= rec["at_recommended"]["order_limit_cents"]  # the card's own figures are consistent
155	    up = rec["next_amount_up"]
156	    assert up["amount_cents"] == chosen.amount_cents + AMOUNT_STEP_CENTS and not up["passes"]
157	    for sc in s["structures"]["inst_90_2000000"]["views"]["event_adjusted"]["scenarios"]:
158	        assert "collections" in sc and "economics" in sc  # per-branch collections and economics
