1	"""Parse acquired source documents into intact sections and structured tables.
2	
3	No fixed-size chunking: a section runs from one detected heading to the next and keeps its heading
4	path. Financial tables keep their rows, header rows, caption, surrounding context and unit hints.
5	Only an oversized section is divided, at block boundaries, into numbered parts that keep the heading.
6	"""

22	BLOCK_TAGS = {"p", "div", "table", "ul", "ol", "li", "h1", "h2", "h3", "h4", "h5", "h6",
23	              "section", "article", "blockquote", "center", "tr", "td", "th"}
24	SKIP_TAGS = {"script", "style", "head", "title", "noscript", "ix:header", "svg", "button", "nav"}
25	NUMBER = re.compile(r"\(?\$?\s?\d[\d,]*(\.\d+)?\)?%?")
26	YEARLIKE = re.compile(r"^(19|20)\d\d$|^[A-Z][a-z]+\.? \d{1,2},? (19|20)\d\d$")
27	PAGE_FURNITURE = re.compile(r"^(F-\d{1,3}|\d{1,3}|Table of Contents|Page \d+ of \d+)$|^Field: ", re.I)
28	ARTICLE_HEADING = re.compile(r"^\d{1,2}\.\s+[A-Z][^.:]{2,80}$")  # "4. Repayment of the Loan; Authorizations"
29	SUBCLAUSE = re.compile(r"^\d+(\.\d+)+\.?\s")  # "3.2. Business Account. You agree ..." is body text
30	STATEMENT_HEADING = re.compile(
31	    r"(Consolidated|Condensed).{0,40}(Balance Sheets?|Statements? of)", re.I)
32	NOTE_HEADING = re.compile(r"^Note \d+\s*[—–-]\s*.{3,100}$")
33	UNIT_HINT = re.compile(r"in (thousands|millions)|\(\s*in [^)]*\)|U\.?S\.? dollars|USD|\$|%", re.I)

36	def norm(text: str) -> str:
37	    return re.sub(r"\s+", " ", text.replace("\xa0", " ")).strip()

40	@dataclass
41	class Block:
42	    kind: str  # "para" | "heading" | "table"
43	    text: str = ""
44	    level: int = 0
45	    rows: list[list[str]] = field(default_factory=list)
46	    header_rows: int = 0
47	    page: int | None = None

52	def _hidden(tag: Tag) -> bool:
53	    return "display:none" in (tag.get("style") or "").replace(" ", "").lower()
54	
55	
56	def _is_bold(node: NavigableString, stop: Tag) -> bool:
57	    for parent in node.parents:
58	        if parent.name in ("b", "strong"):
59	            return True
60	        m = re.search(r"font-weight:\s*(\w+)", (parent.get("style") or "").lower())
61	        if m:
62	            return m.group(1) in ("bold", "bolder", "700", "800", "900")
63	        if parent is stop:
64	            return False
65	    return False
66	
67	
68	def _classify(text: str, bold_share: float) -> Block | None:
69	    if not text or PAGE_FURNITURE.match(text):
70	        return None
71	    if NOTE_HEADING.match(text):  # includes "Note 11 — Notes Payable (cont.)" after a page break
72	        return Block("heading", text.removesuffix("(cont.)").strip(), level=2)
73	    if ARTICLE_HEADING.match(text):
74	        return Block("heading", text, level=2)
75	    if SUBCLAUSE.match(text):
76	        return Block("para", text)
77	    letters = re.sub(r"[^A-Za-z]", "", text)
78	    if letters and len(text) <= 300 and bold_share >= 0.9:
79	        top = (letters.isupper() and len(text) <= 120) or STATEMENT_HEADING.search(text)
80	        level = 1 if top else 2
81	        return Block("heading", text, level=level)
82	    return Block("para", text)

85	def _paragraph(tag: Tag) -> Block | None:
86	    strings = [s for s in tag.find_all(string=True) if norm(s)]
87	    total = sum(len(norm(s)) for s in strings)
88	    bold = sum(len(norm(s)) for s in strings if _is_bold(s, tag))
89	    return _classify(norm(" ".join(strings)), bold / total if total else 0.0)

92	def _table(tag: Tag) -> list[Block]:
93	    grid: list[list[str]] = []
94	    for tr in tag.find_all("tr"):
95	        row: list[str] = []
96	        for cell in tr.find_all(["td", "th"], recursive=False):
97	            span = str(cell.get("colspan") or "1")
98	            row.append(norm(cell.get_text(" ")))
99	            row.extend([""] * (int(span) - 1 if span.isdigit() else 0))
100	        grid.append(row)
101	    width = max((len(r) for r in grid), default=0)
102	    grid = [r + [""] * (width - len(r)) for r in grid]
103	
104	    # Merge currency and closing-paren/percent cells into the number they belong to.
105	    for r in grid:
106	        for i, v in enumerate(r):
107	            if v in ("$", "US$") and i + 1 < width:
108	                j = next((k for k in range(i + 1, width) if r[k]), None)
109	                if j is not None:
110	                    r[j], r[i] = f"{v}{r[j]}", ""
111	            elif v in (")", "%", ")%") and i > 0:
112	                j = next((k for k in range(i - 1, -1, -1) if r[k]), None)
113	                if j is not None:
114	                    r[j], r[i] = r[j] + v, ""
115	    keep = [c for c in range(width) if any(r[c] for r in grid)]
116	    rows = [[r[c] for c in keep] for r in grid if any(r[c] for c in keep)]
117	
118	    data = len(rows) >= 2 and len(keep) >= 2 and any(
119	        NUMBER.fullmatch(v) and not YEARLIKE.match(v) for r in rows for v in r[1:])
120	    if not data:  # layout table (bullets, signature blocks, footnote grids): keep as paragraphs
121	        texts = (" ".join(v for v in r if v) for r in rows)
122	        return [b for t in texts if (b := _classify(t, 0.0))]
123	
124	    header = 0
125	    for r in rows:
126	        numeric = [v for v in r[1:] if v and NUMBER.fullmatch(v) and not YEARLIKE.match(v)]
127	        if r[0] and numeric:
128	            break
129	        header += 1
130	    header = min(header, len(rows))
131	
132	    # A spanning header ("June 30, 2024" over the $ and amount cells) lands one column away from
133	    # its values. Merge two adjacent value columns when no row fills both and one of them holds
134	    # only header text. The label column is never merged.
135	    c = 1
136	    while c + 1 < len(rows[0]):
137	        a = [r[c] for r in rows]
138	        b = [r[c + 1] for r in rows]
139	        disjoint = not any(x and y for x, y in zip(a, b, strict=True))
140	        header_only = not any(a[header:]) or not any(b[header:])
141	        if disjoint and header_only:
142	            for r in rows:
143	                r[c] = r[c] or r[c + 1]
144	                del r[c + 1]
145	        else:
146	            c += 1
147	    return [Block("table", rows=rows, header_rows=header)]
148	
149	
150	def _walk(node: Tag, out: list[Block]) -> None:
151	    for child in node.children:
152	        if isinstance(child, NavigableString):
153	            if norm(child) and node.name not in SKIP_TAGS:
154	                out.append(Block("para", norm(child)))
155	            continue
156	        if not isinstance(child, Tag) or child.name in SKIP_TAGS or _hidden(child):
157	            continue
158	        if child.name == "table":
159	            out.extend(_table(child))
160	        elif re.fullmatch(r"h[1-6]", child.name):
161	            if text := norm(child.get_text(" ")):
162	                out.append(Block("heading", text, level=1 if child.name in ("h1", "h2") else 2))
163	        elif child.name in BLOCK_TAGS and not child.find(BLOCK_TAGS):
164	            if block := _paragraph(child):
165	                out.append(block)
166	        else:
167	            _walk(child, out)
