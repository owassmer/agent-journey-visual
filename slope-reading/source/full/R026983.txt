1	"""Scenario engine: how each Slope financing structure performs, bank-only and along the disputes' paths.
2	
3	Views (matched pair; common inputs identical, only the dispute paths differ):
4	  - bank_only: the connected-bank projection and the financing action.
5	  - event_adjusted: the same, plus one path from each dispute (the generic dispute model gives each dispute its
6	    root-to-leaf paths with conditional weights; distinct disputes are combined as independent, labelled). Each path's
7	    cash is placed two ways: `stress` (outflows and locks at the window start at the high amount; inflows at the
8	    window end at the low amount) and `central` (window midpoint, midpoint amount).
9	
10	The financing action: Slope pays the supplier up to the financed amount on the funding date; the borrower pays any
11	remainder of the invoice itself on its due date (all of it if Slope declines), then repays Slope on the structure's
12	schedule. Collections (declared rule): ACH autopay takes each scheduled payment when available cash covers it;
13	otherwise it takes the available cash and retries the rest at the next month-end (catch-up).
14	
15	Weighted (expected) figures use path weights (products of Jev's conditional judgments, labelled model judgment).
16	Every conditional result is kept beside them. Money is integer cents; rates are Decimal.
17	"""

67	def _place(amount, window_start: date, window_end: date, kind: str, placement: str) -> tuple[date, int]:
68	    lo = amount.value if amount.value is not None else amount.lower
69	    hi = amount.value if amount.value is not None else amount.upper
70	    if placement == "stress":
71	        return (window_start, hi) if kind in ("outflow", "lock") else (window_end, lo)
72	    mid = window_start + (window_end - window_start) / 2
73	    return mid, (lo + hi) // 2

76	def _product(weights: list[int]) -> Decimal:
77	    p = Decimal(1)
78	    for w in weights:
79	        p *= Decimal(w) / BPS
80	    return p
81	
82	
83	def _combination_weights(combos: list[tuple[DisputePath, ...]]) -> list[int | None]:
84	    """Products of the disputes' path weights (combined as independent), in basis points summing to exactly 10,000."""
85	    if any(p.weight_bps is None for combo in combos for p in combo):
86	        return [None] * len(combos)
87	    raw = [_product([p.weight_bps for p in combo]) * BPS for combo in combos]
88	    out = [int(r.to_integral_value()) for r in raw]
89	    out[out.index(max(out))] += 10_000 - sum(out)
90	    return out

93	def scenarios(disputes: list[list[DisputePath]]) -> list[Scenario]:
94	    views = [Scenario("bank_only", (), (), 10_000, "central")]
95	    if not disputes:
96	        return views
97	    combos = list(itertools.product(*disputes))
98	    for combo, w in zip(combos, _combination_weights(combos), strict=True):
99	        cash = tuple(c for p in combo for c in p.cash)
100	        for placement in (("stress", "central") if cash else ("central",)):
101	            views.append(Scenario("event_adjusted", tuple(p.path_id for p in combo),
102	                                  tuple(" → ".join(p.labels) for p in combo), w, placement, cash))
103	    return views
