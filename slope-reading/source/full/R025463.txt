70	def _obs_view(o: SemanticObservation) -> dict[str, Any]:
71	    view = {"observation_id": o.observation_id, "question": registry_question(o.question_id)["question"],
72	            "question_id": o.question_id, "answer": o.answer, "meaning": meaning(o.question_id, o.answer),
73	            "ambiguous": is_ambiguous(o)}
74	    if o.probabilities:
75	        view["judgment_distribution"] = {k: round(v, 2) for k, v in o.probabilities.items()}
76	    if o.noul_value is not None:
77	        view["judgment_value"] = round(o.noul_value, 2)
78	    return view
