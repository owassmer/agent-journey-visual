1	"""Scoped MCP tools for one investigation run (spec §7; agent_config tools section).

3	Every handler is bound to one run: its event store, its admissible snapshot, its locked inputs and (in
4	the agent-plus-Jev arm) its semantic layer. Arguments are object IDs, queries and agent-authored
5	propositions; no tool accepts a path, database, snapshot, cutoff, raw Jev state or question ID.
6	
7	Host rules enforced here:
8	- search results are always all returned; in the Jev arm each carries its screen label;
9	- a finding cites verbatim quotes; in the Jev arm every finding gets finding_check before resolution;
10	- accepting a finding requires a disposition for each linked observation, a reasoned override for
11	  any unsettled answer, and never accepts an ambiguous entity_scope answer;
12	- effects are validated against accepted findings and re-verified spans; Jev never activates one;
13	- sensitivities are deterministic finance-engine calls over explicit inputs.
14	"""

103	def _obs_view(o: SemanticObservation) -> dict[str, Any]:
104	    view = {"observation_id": o.observation_id, "question": registry_question(o.question_id)["question"],
105	            "question_id": o.question_id, "answer": o.answer, "meaning": meaning(o.question_id, o.answer),
106	            "ambiguous": is_ambiguous(o)}
107	    if o.probabilities:
108	        view["judgment_distribution"] = {k: round(v, 2) for k, v in o.probabilities.items()}
109	    if o.noul_value is not None:
110	        view["judgment_value"] = round(o.noul_value, 2)
111	    return view
