26	def decimal(value):
27	    """Reject binary floating point and non-finite monetary inputs."""
28	    if isinstance(value, bool) or isinstance(value, float):
29	        raise TypeError("Use Decimal, decimal string, or integer, never float/bool")
30	    result = value if isinstance(value, D) else D(value)
31	    if not result.is_finite():
32	        raise ValueError("Input must be finite")
33	    return result
34	
35	
36	def nonnegative(value, name):
37	    result = decimal(value)
38	    if result < ZERO:
39	        raise ValueError(f"{name} must be nonnegative")
40	    return result
41	
42	
43	def money(value):
44	    return format(decimal(value).quantize(CENT, rounding=ROUND_HALF_UP), ".2f")

47	@dataclass(frozen=True)
48	class MerchantTerms:
49	    advance: D = D("370000")
50	    fixed_cost: D = D("48100")
51	    total_payment: D = D("418100")
52	    remittance_rate: D = D("0.25")
53	
54	    def __post_init__(self):
55	        for key in ("advance", "fixed_cost", "total_payment", "remittance_rate"):
56	            object.__setattr__(self, key, nonnegative(getattr(self, key), key))
57	        if self.advance + self.fixed_cost != self.total_payment:
58	            raise ValueError("Advance plus fixed cost must equal total payment")
59	        if not ZERO < self.remittance_rate <= D("1"):
60	            raise ValueError("Remittance rate must be in (0,1]")
61	
62	    def cumulative_floor(self, month):
63	        fractions = {6: D("0.30"), 12: D("0.60"), 18: D("1")}
64	        if month not in fractions:
65	            raise ValueError("Only contract checkpoints 6, 12 and 18 months are defined")
66	        return self.total_payment * fractions[month]
67	
68	    def qualifying_credit_threshold(self, month):
69	        """Credits needed if the floor is met entirely by the 25% remittance."""
70	        return self.cumulative_floor(month) / self.remittance_rate
71	
72	    def cumulative_shortfall_only(self, month, remittances_received, manual_payments_received):
73	        """Necessary cumulative check only; NOT sufficient for the 12-month test."""
74	        daily = nonnegative(remittances_received, "remittances_received")
75	        manual = nonnegative(manual_payments_received, "manual_payments_received")
76	        if daily + manual > self.total_payment:
77	            raise ValueError("Payments cannot exceed the contractual total in this calculator")
78	        return max(ZERO, self.cumulative_floor(month) - daily - manual)
79	
80	    def window_top_up(self, month, daily_payments_in_window,
81	                      accepted_manual_credit_for_window, total_received_to_date):
82	        """Six-month window shortfall, capped at remaining total obligation.
83	
84	        Section 4.2.1(b) measures Daily Payments within each six-month window.
85	        Manual credit must be confirmed for this window; no automatic carry of
86	        first-window overpayments or unverified/reversed manual payments.
87	        Earlier unmet obligations are not erased by this current-window test.
88	        """
89	        if month not in (6, 12):
90	            raise ValueError("Window test is defined only for months 6 and 12")
91	        daily = nonnegative(daily_payments_in_window, "daily_payments_in_window")
92	        manual = nonnegative(accepted_manual_credit_for_window, "accepted_manual_credit_for_window")
93	        received = nonnegative(total_received_to_date, "total_received_to_date")
94	        if daily + manual > received or received > self.total_payment:
95	            raise ValueError("Window credits must fit within total lender receipts and contractual total")
96	        shortfall = max(ZERO, self.total_payment * D("0.30") - daily - manual)
97	        return min(self.total_payment - received, shortfall)

100	@dataclass(frozen=True)
101	class ConditionalBudget:
102	    """Single-horizon identity; None means unknown, never an assumed zero.
103	
104	    Opening cash is the August 12 closing cash anchor. Future flows start after
105	    that date. Operating outflows EXCLUDE supplier settlements, debt service and
106	    reserve. Debt service EXCLUDES the supplier settlements already in H2 dues.
107	    Cash collections are before separately modeled debt remittances. Inflows
108	    must exclude any hypothetical new facility being evaluated.
109	    """
110	    opening_cash: D = D("2000000")  # Approximate management-disclosed amount.
111	    h2_settlements_remaining_at_june30: D = D("2600000")
112	    unavailable_portion_of_opening_cash: D | None = None
113	    settlements_paid_july1_through_august12: D | None = None
114	    future_customer_cash_collections: D | None = None
115	    future_committed_financing_draws: D | None = None
116	    future_other_confirmed_cash_inflows: D | None = None
117	    future_operating_cash_outflows_excluding_settlements_and_debt: D | None = None
118	    future_other_debt_service_excluding_settlements: D | None = None
119	    required_cash_reserve: D | None = None
120	
121	    def __post_init__(self):
122	        for name, value in vars(self).items():
123	            if value is not None:
124	                object.__setattr__(self, name, nonnegative(value, name))
125	        paid = self.settlements_paid_july1_through_august12
126	        if paid is not None and paid > self.h2_settlements_remaining_at_june30:
127	            raise ValueError("Already-paid amount exceeds this specific H2 settlement bucket")
128	        unavailable = self.unavailable_portion_of_opening_cash
129	        if unavailable is not None and unavailable > self.opening_cash:
130	            raise ValueError("Unavailable portion exceeds reported opening cash")
131	
132	    def unknown_fields(self):
133	        return [name for name, value in vars(self).items() if value is None]
134	
135	    def remaining_settlement_cash(self):
136	        if self.settlements_paid_july1_through_august12 is None:
137	            return None
138	        return (self.h2_settlements_remaining_at_june30
139	                - self.settlements_paid_july1_through_august12)
140	
141	    def horizon_residual_before_new_facility(self):
142	        """Conditional cash residual, NOT a maximum loan or an observed gap."""
143	        if self.unknown_fields():
144	            return None
145	        return (
146	            self.opening_cash
147	            - self.unavailable_portion_of_opening_cash
148	            + self.future_customer_cash_collections
149	            + self.future_committed_financing_draws
150	            + self.future_other_confirmed_cash_inflows
151	            - self.remaining_settlement_cash()
152	            - self.future_operating_cash_outflows_excluding_settlements_and_debt
153	            - self.future_other_debt_service_excluding_settlements
154	            - self.required_cash_reserve
155	        )

158	@dataclass(frozen=True)
159	class CashReceipt:
160	    day: int
161	    amount: D
162	
163	    def __post_init__(self):
164	        if isinstance(self.day, bool) or not isinstance(self.day, int) or self.day < 0:
165	            raise ValueError("Receipt day must be a nonnegative integer")
166	        object.__setattr__(self, "amount", nonnegative(self.amount, "receipt amount"))
167	
168	
169	def delay_receipts(receipts, days):
170	    if isinstance(days, bool) or not isinstance(days, int) or days < 0:
171	        raise ValueError("Delay must be a nonnegative integer")
172	    return tuple(CashReceipt(r.day + days, r.amount) for r in receipts)
173	
174	
175	def present_value(receipts, effective_annual_discount_rate):
176	    """Discount from day zero, effective annual rate, ACT/365 convention."""
177	    rate = nonnegative(effective_annual_discount_rate, "discount rate")
178	    with localcontext() as ctx:
179	        ctx.prec = 40
180	        return sum((r.amount / ctx.power(D("1") + rate, D(r.day) / D("365"))
181	                    for r in receipts), ZERO)

184	def principal_exposure_dollar_days(principal_only_receipts):
185	    """Principal-only fixture: integral of outstanding principal until repaid.
186	
187	    All principal is advanced at day zero and completely returned in the given
188	    receipts. Do not apply this to mixed principal/fee payments without an
189	    explicit allocation rule. No such rule is inferred for the Synergy loan.
190	    """
191	    return sum((r.amount * D(r.day) for r in principal_only_receipts), ZERO)
