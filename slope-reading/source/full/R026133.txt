331	def test_covered_claims_are_checked(make_ctx):
332	    from app.domain.investigation import InventoryItem
333	    ctx = make_ctx(answers={"coverage_supported": "not_covered"}, name="coverage")
334	    note11 = next(h for h in ctx.evidence.search("December 28, 2023 settlement 802,445") if h["kind"] == "section")["id"]
335	    ctx.run.put("inventory_loaded", InventoryItem(item_id="inv_001", section_ids=(note11,), source_id="synergy_s1a_20240813",
336	                                                 heading_path=("Note 11",), kind="accounting_item_from_a_matter", signal=0.9))
337	    _, fid = _accepted_settlement_finding(ctx)
338	    entry = {"item_id": "inv_001", "disposition": "covered_by_findings", "finding_ids": [fid]}
339	    out = call(T.account_for_items, ctx, {"items": [entry]})
340	    assert out["accounted"] == [] and "different matter" in out["rejected"][0]["reason"]
341	    assert ctx.run.get("inventory", "inv_001").status == "open"
342	    # no free-text override of a coverage gap
343	    still = call(T.account_for_items, ctx, {"items": [{**entry, "override_reason": "The schedule is the matter; the gain is elsewhere."}]})
344	    assert still["open"] == 1
345	    failed = next(o.observation_id for o in ctx.run.graph["observations"].values() if o.question_id == "coverage_supported")
346	    missing = "The accounting gain on the settlement is not accounted for by any finding."
347	    bad = call(T.account_for_items, ctx, {"items": [{"item_id": "inv_001", "disposition": "escalate", "observation_id": "obs_999",
348	                                                     "missing": missing}]})
349	    assert "failed host check" in bad["rejected"][0]["reason"]
350	    call(T.account_for_items, ctx, {"items": [{"item_id": "inv_001", "disposition": "escalate", "observation_id": failed,
351	                                               "missing": missing}]})
352	    assert ctx.run.get("inventory", "inv_001").status == "disputed"
353	    call(T.submit_packet, ctx, {"summary": "s", "conclusion": "The settlement loan requires future payments."})
354	    assert not ctx.incomplete_reasons  # an escalated item goes to the reviewer's checklist, not straight to INCOMPLETE
355	    submitted = next(e.payload for e in ctx.run.events if e.kind == "packet_submitted")
356	    assert [d["id"] for d in submitted["disputes"] if d["for"] == "reviewer_checklist"] == ["inv_001"]

360	    from app.domain.investigation import InventoryItem
361	    ctx = make_ctx(answers={"adds_matter": "adds_item", "decision_relevance": "could_change"}, name="dup")
